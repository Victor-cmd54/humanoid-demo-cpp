cmake_minimum_required(VERSION 3.20)
project(humanoid_demo_cpp LANGUAGES CXX)
add_executable(humanoid_demo_cpp humanoid_demo.cpp)
target_compile_features(humanoid_demo_cpp PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(humanoid_demo_cpp PRIVATE /W4 /permissive-)
else()
  target_compile_options(humanoid_demo_cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
