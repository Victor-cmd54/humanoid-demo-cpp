cmake_minimum_required(VERSION 3.20)
project(humanoid_demo_cpp LANGUAGES CXX)
add_executable(humanoid_demo_cpp humanoid_demo.cpp)
target_compile_features(humanoid_demo_cpp PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(humanoid_demo_cpp PRIVATE /W4 /permissive-)
else()
  target_compile_options(humanoid_demo_cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(section_1_4_locomotion src/section_1_4_locomotion.cpp)
target_compile_features(section_1_4_locomotion PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(section_1_4_locomotion PRIVATE /W4 /permissive-)
else()
  target_compile_options(section_1_4_locomotion PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_test(NAME section_1_4_locomotion_demo COMMAND section_1_4_locomotion)
